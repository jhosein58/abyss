Arena_alloc :: (a &Arena, s u64) &u8 {
    aligned_size := align_up(s)
    curr := (*a).current

    if (*curr).offset + aligned_size > (*curr).cap {
        new_cap := (*a).default_chunk_size

        if aligned_size > new_cap {
            new_cap = aligned_size
        }

        new_chunk := ArenaChunk_new(new_cap, curr as &u8)
        (*a).current = new_chunk
        curr = new_chunk
    }

    ptr := (*curr).memory + (*curr).offset
    (*curr).offset = (*curr).offset + aligned_size

    ptr
}

Arena_reset :: (a &Arena) {
    curr := (*a).current
    while (curr as &u8) != (0 as &u8) {
        (*curr).offset = 0
        curr = (*curr).next_chunk as &ArenaChunk
    }
}


Arena_realloc :: (a &Arena, old_ptr &u8, old_size u64, new_size u64) &u8 {

        old_ptr

        aligned_old := align_up(old_size)
        aligned_new := align_up(new_size)
        curr := (*a).current

        if old_ptr == last_alloc_ptr {
            diff := aligned_new - aligned_old

            if (*curr).offset + diff <= (*curr).cap {
                (*curr).offset = (*curr).offset + diff
                old_ptr
            } else {
                new_ptr := Arena_alloc(a, new_size)

                new_ptr

        } else {

            new_ptr
        }
